// 1. Mutex chống bão gọi lặp /auth/refresh khi nhiều API cùng nhận 401
let refreshPromise = null;
async function getFreshAccessToken(refreshUrl) {
  if (!refreshPromise) {
    refreshPromise = fetch(refreshUrl, { method: 'POST' })
      .then((r) => r.json())
      .then((d) => d.accessToken)
      .finally(() => { refreshPromise = null; });
  }
  return refreshPromise;
}

// 2. Wrapper Fetch chống bão 429/503, Timeout AbortController, 401 Mutex & Idempotency
export async function fetchResilient(url, options = {}) {
  const { timeoutMs = 5000, maxRetries = 2, idempotent = false, refreshUrl = '', ...fetchOpts } = options;
  const headers = new Headers(fetchOpts.headers || {});
  if (idempotent && !headers.has('X-Idempotency-Key')) {
    headers.set('X-Idempotency-Key', crypto.randomUUID());
  }

  let lastError;
  for (let attempt = 0; attempt <= maxRetries; attempt++) {
    const controller = new AbortController();
    const timer = setTimeout(() => controller.abort(), timeoutMs);
    try {
      let res = await fetch(url, { ...fetchOpts, headers, signal: controller.signal });
      clearTimeout(timer);

      if (res.status === 401 && refreshUrl && attempt === 0) {
        const newToken = await getFreshAccessToken(refreshUrl);
        headers.set('Authorization', `Bearer ${newToken}`);
        continue;
      }

      if ([429, 502, 503, 504].includes(res.status) && attempt < maxRetries) {
        const retryAfter = Number(res.headers.get('Retry-After')) || 0;
        await new Promise((r) => setTimeout(r, retryAfter > 0 ? retryAfter * 1000 : 350 * Math.pow(2, attempt)));
        continue;
      }

      const rawText = await res.text();
      let data;
      try { data = rawText ? JSON.parse(rawText) : null; }
      catch { throw new Error(`MALFORMED_JSON_RESPONSE (HTTP ${res.status})`); }

      if (!res.ok) throw new Error(data?.message || `HTTP_ERROR_${res.status}`);
      return data;
    } catch (err) {
      clearTimeout(timer);
      lastError = err.name === 'AbortError' ? new Error(`REQUEST_TIMEOUT_${timeoutMs}MS`) : err;
      if (attempt < maxRetries) await new Promise((r) => setTimeout(r, 300 * Math.pow(2, attempt)));
    }
  }
  throw lastError;
}
